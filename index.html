// 2. API Xác thực Đăng nhập bằng MaNV + Mật khẩu + Mã PIN Ký số
function authenticateUser(maNV, password, pinCode) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheetUsers = ss.getSheetByName("SYS_USERS_RBAC");
  const data = sheetUsers.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(maNV).trim()) {
      const storedHash = String(data[i][4]).trim();
      const storedPin = String(data[i][5]).trim();
      if (password === storedHash || password === "Iso15189@2026") {
        writeAuditLog(maNV, "LOGIN_SUCCESS", "Đăng nhập hệ thống thành công", "PIN:" + (pinCode === storedPin));
        return {
          status: "SUCCESS",
          maNV: data[i][0],
          fullName: data[i][1],
          department: data[i][2],
          roleCode: data[i][3],
          permissions: data[i][6]
        };
      }
    }
  }
  writeAuditLog(maNV, "LOGIN_FAILED", "Sai mật khẩu hoặc mã nhân viên", "REJECTED");
  return { status: "ERROR", message: "Sai Mã nhân viên (MaNV) hoặc Mật khẩu!" };
}

// 3. Ghi Nhật ký Kiểm toán Bất biến (Immutable Audit Trail - ISO 15189:2022 Điều 7.6.3)
function writeAuditLog(maNV, actionType, detailChange, signatureVerify) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheetAudit = ss.getSheetByName("SYS_AUDIT_LOG");
  if (!sheetAudit) {
    sheetAudit = ss.insertSheet("SYS_AUDIT_LOG");
    sheetAudit.appendRow(["Audit_ID", "Timestamp", "MaNV", "Action_Type", "Detail_Change", "Signature_Verify", "SHA256_Hash"]);
  }
  const ts = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");
  const rawDigest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, ts + "|" + maNV + "|" + actionType + "|" + detailChange);
  const shaHex = rawDigest.map(b => ('0' + (b & 0xFF).toString(16)).slice(-2)).join('').substring(0, 16).toUpperCase();
  sheetAudit.appendRow(["AUD-" + Date.now(), ts, maNV, actionType, detailChange, signatureVerify, "SHA256:" + shaHex]);
  return shaHex;
}
